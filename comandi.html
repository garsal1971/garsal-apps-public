SETUP
Redirect URL su Supabase — senza questo il login non torna nell'app
Supabase Dashboard → Authentication → URL Configuration → Redirect URLs → aggiungere https://garsal.men/oauth-callback-native.html.
Non garsalnative://oauth: quello è lo schema con cui la pagina-ponte rilancia l'app, e Supabase non lo vede mai. Chrome blocca il salto automatico da una pagina di login a uno schema custom, quindi il login deve atterrare su una pagina https che rimanda all'app col tap dell'utente — è lo stesso giro che AppSfera Web fa da oauth-callback.html.
Se l'elenco ha già una regola con * su garsal.men è coperto così com'è; se manca, il login finisce sul sito e non torna mai nell'app.

🔐 Smart Block — Setup completo (una-tantum)
SQL
Smart Block — DDL + Device Token
Eseguire una sola volta nel SQL Editor di Supabase (prod).
Aggiunge 'smart_block' al CHECK di cm_notification_rules.channel, le RLS per il client Android, il device token e la funzione di lettura per l'app.
una tantum supabase prod 
-- ── 1. cm_notification_rules.channel: aggiunge 'smart_block' ──────
ALTER TABLE cm_notification_rules
    DROP CONSTRAINT IF EXISTS cm_notification_rules_channel_check;

ALTER TABLE cm_notification_rules
    ADD CONSTRAINT cm_notification_rules_channel_check
        CHECK (channel IN ('telegram', 'browser', 'push', 'smart_block'));

-- ── 2. cm_notification_queue: RLS per client Android (anon key) ────
DROP POLICY IF EXISTS "smart_block_anon_select" ON cm_notification_queue;
DROP POLICY IF EXISTS "smart_block_anon_update" ON cm_notification_queue;

CREATE POLICY "smart_block_anon_select"
    ON cm_notification_queue FOR SELECT TO anon
    USING (channel = 'smart_block');

CREATE POLICY "smart_block_anon_update"
    ON cm_notification_queue FOR UPDATE TO anon
    USING  (channel = 'smart_block')
    WITH CHECK (channel = 'smart_block' AND status IN ('sent', 'failed'));

-- ── 3. Device token in cm_user_notification_settings ───────────────
ALTER TABLE cm_user_notification_settings
    ADD COLUMN IF NOT EXISTS smart_block_device_token text;

INSERT INTO cm_user_notification_settings (user_id, smart_block_device_token)
SELECT id, gen_random_uuid()
FROM   auth.users
LIMIT  1
ON CONFLICT (user_id) DO UPDATE
SET smart_block_device_token = COALESCE(
    cm_user_notification_settings.smart_block_device_token,
    EXCLUDED.smart_block_device_token
);

-- ── 4. Funzione read-only per app Android (anon key) ───────────────
CREATE OR REPLACE FUNCTION public.get_smart_block_token()
RETURNS text LANGUAGE sql SECURITY DEFINER SET search_path = public AS $$
    SELECT smart_block_device_token FROM cm_user_notification_settings LIMIT 1;
$$;

GRANT EXECUTE ON FUNCTION public.get_smart_block_token()    TO anon;
GRANT EXECUTE ON FUNCTION public.task_complete(uuid, date)  TO anon;

-- Mostra il token generato
SELECT smart_block_device_token FROM cm_user_notification_settings;

-- Token impostato
SELECT user_id, smart_block_device_token FROM cm_user_notification_settings;

-- Regole smart_block (device_token nel JSON)
SELECT id, entity_title, reminder_presets->>'device_token' AS token,
       reminder_presets->>'smart_block_fire_at' AS fire_at
FROM   cm_notification_rules
WHERE  channel = 'smart_block';

-- Coda smart_block per stato
SELECT status, count(*)
FROM   cm_notification_queue
WHERE  channel = 'smart_block'
GROUP  BY status;

-- Funzioni anon accessibili
SELECT routine_name FROM information_schema.routines
WHERE  routine_name IN ('get_smart_block_token', 'task_complete')
AND    routine_schema = 'public';
